add_namespace = hsm_ver_uni

# Refusal is first: an unattended human event never defaults to surrender.
country_event = {
	id = hsm_ver_uni.10
	title = hsm_ver_uni.10.t
	desc = hsm_ver_uni.10.d
	picture = GFX_report_event_generic_parliament
	is_triggered_only = yes timeout_days = 30
	trigger = {
		HSM_VER_UNI_house = yes NOT = { tag = ERI }
		has_country_flag = HSM_VER_UNI_invite_pending_ERI
		ERI = { has_country_flag = HSM_VER_UNI_leader NOT = { has_country_flag = HSM_VER_UNI_closed has_country_flag = HSM_VER_UNI_founded } }
	}
	option = {
		name = hsm_ver_uni.decline
		custom_effect_tooltip = HSM_VER_EVT_decline_union_tt
		ai_chance = { factor = 100 modifier = { factor = 0 HSM_VER_UNI_invitation_valid_ERI = yes } }
		hidden_effect = {
			if = { limit = { HSM_VER_UNI_invitation_valid_ERI = yes } HSM_VER_UNI_withdraw_ERI = yes }
		}
	}
	option = {
		name = hsm_ver_uni.participate
		trigger = { HSM_VER_UNI_invitation_valid_ERI = yes }
		ai_chance = { factor = 100 }
		custom_effect_tooltip = HSM_VER_UNI_participation_tt
		hidden_effect = { HSM_VER_UNI_accept_invitation_ERI = yes }
	}
}
