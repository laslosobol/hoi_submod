    id = hsm_bar_diplomacy.90
    title = HSM_BAR_dip_failed_t
    desc = HSM_BAR_dip_failed_d
    picture = GFX_event_pony_soldiers_sunset
    is_triggered_only = yes
    trigger = {
        OR = {
            AND = { tag = WIT HSM_BAR_dip_WIT_valid = yes BAR = { has_country_flag = HSM_BAR_dip_WIT_rejected NOT = { has_country_flag = HSM_BAR_WIT_agreement } } }
            AND = { tag = ZES HSM_BAR_dip_ZES_valid = yes BAR = { has_country_flag = HSM_BAR_dip_ZES_rejected NOT = { has_country_flag = HSM_BAR_ZES_agreement } } }
            AND = { tag = OLV HSM_BAR_dip_OLV_valid = yes BAR = { has_country_flag = HSM_BAR_dip_OLV_rejected NOT = { has_country_flag = HSM_BAR_OLV_agreement } } }
        }
    }
    option = {
        name = HSM_BAR_dip_enforce
        ai_chance = { base = 80 modifier = { factor = 0.25 has_war = yes } }
        custom_effect_tooltip = HSM_BAR_dip_enforce_tt
        if = {
            limit = { tag = WIT HSM_BAR_dip_WIT_valid = yes BAR = { has_country_flag = HSM_BAR_dip_WIT_rejected NOT = { has_country_flag = HSM_BAR_WIT_agreement } } }
            create_wargoal = { type = annex_everything target = BAR }
            if = { limit = { can_declare_war_on = BAR } declare_war_on = { target = BAR type = annex_everything } }
        }
        if = {
            limit = { tag = ZES HSM_BAR_dip_ZES_valid = yes BAR = { has_country_flag = HSM_BAR_dip_ZES_rejected NOT = { has_country_flag = HSM_BAR_ZES_agreement } } }
            # Retain Kasa's native war/victory continuation if talks fail.
            if = {
                limit = { has_country_flag = HSM_BAR_dip_yrmthu_ready }
                set_country_flag = ZES_yrmthu_barrad_war
                BAR = { add_ideas = ZES_barrad_partisans }
            }
            if = {
                limit = { has_country_flag = HSM_BAR_dip_altu_ready }
                set_country_flag = ZES_altu_barrad_war
                if = {
                    limit = { has_country_flag = ZES_barrad_partisan_flag }
                    BAR = { add_ideas = ZES_barrad_partisans }
                }
                else_if = {
                    limit = { has_country_flag = ZES_barrad_propaganda_flag }
                    add_ideas = ZES_barrad_propaganda
                }
            }
            create_wargoal = { type = annex_everything target = BAR }
            if = { limit = { can_declare_war_on = BAR } declare_war_on = { target = BAR type = annex_everything } }
        }
        if = {
            limit = { tag = OLV HSM_BAR_dip_OLV_valid = yes BAR = { has_country_flag = HSM_BAR_dip_OLV_rejected NOT = { has_country_flag = HSM_BAR_OLV_agreement } } }
            create_wargoal = { type = annex_everything target = BAR }
            if = { limit = { can_declare_war_on = BAR } declare_war_on = { target = BAR type = annex_everything } }
        }
    }
    option = {
        name = HSM_BAR_dip_withdraw
        ai_chance = { base = 20 }
        if = {
            limit = { tag = WIT HSM_BAR_dip_WIT_valid = yes BAR = { has_country_flag = HSM_BAR_dip_WIT_rejected } }
            add_political_power = -50
            remove_wargoal = { type = annex_everything target = BAR }
            set_country_flag = HSM_BAR_dip_settled
            BAR = {
                clr_country_flag = HSM_BAR_dip_WIT_rejected
                clr_country_flag = HSM_BAR_dip_WIT_escalated
                set_country_flag = { flag = HSM_BAR_dip_WIT_cooldown days = 180 }
            }
            
        }
        if = {
            limit = { tag = ZES HSM_BAR_dip_ZES_valid = yes BAR = { has_country_flag = HSM_BAR_dip_ZES_rejected } }
            add_political_power = -50
            remove_wargoal = { type = annex_everything target = BAR }
            set_country_flag = HSM_BAR_dip_settled
            BAR = {
                clr_country_flag = HSM_BAR_dip_ZES_rejected
                clr_country_flag = HSM_BAR_dip_ZES_escalated
                set_country_flag = { flag = HSM_BAR_dip_ZES_cooldown days = 180 }
            }
            HSM_BAR_dip_ZES_progress = yes
        }
        if = {
            limit = { tag = OLV HSM_BAR_dip_OLV_valid = yes BAR = { has_country_flag = HSM_BAR_dip_OLV_rejected } }
            add_political_power = -50
            remove_wargoal = { type = annex_everything target = BAR }
            set_country_flag = HSM_BAR_dip_settled
            BAR = {
                clr_country_flag = HSM_BAR_dip_OLV_rejected
                clr_country_flag = HSM_BAR_dip_OLV_escalated
                set_country_flag = { flag = HSM_BAR_dip_OLV_cooldown days = 180 }
            }
            
        }
    }
