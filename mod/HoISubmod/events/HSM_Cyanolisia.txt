country_event = {
	id = hsm_cyanolisia.1
	immediate = { log = "[GetDateText]: [Root.GetName]: event hsm_cyanolisia.1" }
	title = hsm_cyanolisia.1.t
	desc = hsm_cyanolisia.1.d
	picture = GFX_event_griffon_with_gun
	is_triggered_only = yes

	trigger = {
		tag = CYA
		NOT = { has_country_flag = HSM_CYA_dawnclaw_asylum_resolved }
		GRI = {
			exists = yes
			NOT = {
				has_country_leader = {
					character = GRI_ferdinand_dawnclaw
					ruling_only = yes
				}
			}
		}
	}

	option = {
		name = hsm_cyanolisia.1.a
		ai_chance = { factor = 70 }
		set_country_flag = HSM_CYA_dawnclaw_asylum_resolved
		set_country_flag = HSM_CYA_dawnclaw_hidden_in_court
		set_country_flag = HSM_CYA_dawnclaw_branch_unlocked
		set_country_flag = HSM_CYA_dawnclaw_programs_reworked
		HSM_CYA_install_dawnclaw_programs = yes
		set_variable = { HSM_CYA_dawnclaw_exposure = 1 }
		if = {
			limit = {
				NOT = { has_character = HSM_CYA_ferdinand_dawnclaw }
			}
			recruit_character = HSM_CYA_ferdinand_dawnclaw
		}
		activate_advisor = HSM_CYA_ferdinand_dawnclaw
		set_can_be_fired_in_advisor_role = {
			character = HSM_CYA_ferdinand_dawnclaw
			slot = political_advisor
			value = no
		}
		add_political_power = -25
		add_to_variable = { HSM_CYA_decryption_factor = 0.02 }
		add_to_variable = { HSM_CYA_imperial_drift_defence_factor = 0.02 }
		hidden_effect = {
			GRI = { set_country_flag = HSM_GRI_dawnclaw_hidden_in_cyanolisia }
			load_focus_tree = {
				tree = cyan_original
				keep_completed = yes
			}
			mark_focus_tree_layout_dirty = yes
		}
		custom_effect_tooltip = HSM_CYA_dawnclaw_assistant_appointed_tt
	}

	option = {
		name = hsm_cyanolisia.1.b
		ai_chance = { factor = 30 }
		set_country_flag = HSM_CYA_dawnclaw_asylum_resolved
	}
}
