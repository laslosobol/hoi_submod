add_namespace = hsm_cyanolisia

country_event = {
	id = hsm_cyanolisia.1
	immediate = { log = "[GetDateText]: [Root.GetName]: event hsm_cyanolisia.1" }
	title = hsm_cyanolisia.1.t
	desc = hsm_cyanolisia.1.d
	picture = GFX_event_griffon_with_gun
	is_triggered_only = yes

	trigger = {
		tag = CYA
		NOT = { has_country_flag = HSM_CYA_dawnclaw_asylum_resolved }
		GRI = {
			has_completed_focus = GRI_industry_modern
			NOT = {
				has_country_leader = {
					character = GRI_ferdinand_dawnclaw
					ruling_only = yes
				}
			}
		}
	}

	option = {
		name = hsm_cyanolisia.1.a
		ai_chance = { factor = 70 }
		set_country_flag = HSM_CYA_dawnclaw_asylum_resolved
		set_country_flag = HSM_CYA_dawnclaw_hidden_in_court
		activate_advisor = HSM_CYA_ferdinand_dawnclaw
		set_can_be_fired_in_advisor_role = {
			character = HSM_CYA_ferdinand_dawnclaw
			slot = political_advisor
			value = no
		}
		add_ideas = HSM_CYA_dawnclaw_griffonian_files
		add_political_power = -25
		add_stability = -0.02
		hidden_effect = {
			GRI = { set_country_flag = HSM_GRI_dawnclaw_hidden_in_cyanolisia }
			country_event = { id = hsm_cyanolisia.2 days = 7 }
			country_event = { id = hsm_cyanolisia.10 days = 60 random = 60 }
		}
		custom_effect_tooltip = HSM_CYA_dawnclaw_assistant_appointed_tt
	}

	option = {
		name = hsm_cyanolisia.1.b
		ai_chance = { factor = 30 }
		set_country_flag = HSM_CYA_dawnclaw_asylum_resolved
		add_political_power = 25
	}
}
