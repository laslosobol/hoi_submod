country_event = {
    id = hsm_bar_anomaly.24
    title = hsm_bar_anomaly.24.t
    desc = hsm_bar_anomaly.24.d
    picture = GFX_event_gothic_interior
    is_triggered_only = yes
    immediate = {
        HSM_BAR_anomalies_pulse = yes
        set_variable = { HSM_BAR_case_crown_museum_count = 0 }
        set_variable = { HSM_BAR_case_crown_sealed_count = 0 }
        set_variable = { HSM_BAR_case_crown_leader_count = 0 }
        set_variable = { HSM_BAR_case_crown_missing_count = 0 }
        set_variable = { HSM_BAR_case_crown_vault_count = 0 }
        for_each_scope_loop = {
            array = global.HSM_BAR_case_crowns
            if = {
                limit = { has_country_flag = HSM_BAR_case_crown_known }
                if = { limit = { has_country_flag = HSM_BAR_case_crown_museum } ROOT = { add_to_variable = { HSM_BAR_case_crown_museum_count = 1 } } }
                if = { limit = { has_country_flag = HSM_BAR_case_crown_sealed } ROOT = { add_to_variable = { HSM_BAR_case_crown_sealed_count = 1 } } }
                if = { limit = { has_country_flag = HSM_BAR_case_crown_leader } ROOT = { add_to_variable = { HSM_BAR_case_crown_leader_count = 1 } } }
                if = { limit = { has_country_flag = HSM_BAR_case_crown_missing } ROOT = { add_to_variable = { HSM_BAR_case_crown_missing_count = 1 } } }
                if = {
                    limit = { has_country_flag = HSM_BAR_case_crown_secured var:HSM_BAR_case_crown_site = { is_controlled_by = ROOT } }
                    ROOT = { add_to_variable = { HSM_BAR_case_crown_vault_count = 1 } }
                }
            }
        }
    }
    option = { name = hsm_bar_anomaly.file_report }
}
