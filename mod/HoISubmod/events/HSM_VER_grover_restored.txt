# Focus-triggered adult scenes. The first two revise inherited legacies on GRI only.
country_event = {
	id = hsm_ver_grover_restored.1
	title = hsm_ver_grover_restored.1.t
	desc = {
		trigger = { has_country_flag = HSM_VER_GRO_filtered_education }
		text = hsm_ver_grover_restored.1.d_filtered
	}
	desc = {
		trigger = { NOT = { has_country_flag = HSM_VER_GRO_filtered_education } }
		text = hsm_ver_grover_restored.1.d
	}
	picture = GFX_event_griffenheim
	is_triggered_only = yes
	fire_only_once = yes
	trigger = {
		original_tag = GRI
		has_country_flag = HSM_VER_GRO_restored
		is_subject = no
		has_country_leader = { character = GRI_emperor_grover_vi ruling_only = yes }
		has_completed_focus = HSM_VER_GRO_restored_oath
	}
	option = {
		name = hsm_ver_grover_restored.1.a
		ai_chance = { factor = 70 }
		if = {
			limit = { hidden_trigger = {
				original_tag = GRI has_country_flag = HSM_VER_GRO_restored is_subject = no
				has_country_leader = { character = GRI_emperor_grover_vi ruling_only = yes }
			} }
			remove_ideas = HSM_VER_GRO_managed_court_legacy
			add_ideas = HSM_VER_GRO_open_court_legacy
		}
	}
	option = {
		name = hsm_ver_grover_restored.1.b
		ai_chance = { factor = 30 }
		if = {
			limit = { hidden_trigger = {
				original_tag = GRI has_country_flag = HSM_VER_GRO_restored is_subject = no
				has_country_leader = { character = GRI_emperor_grover_vi ruling_only = yes }
			} }
			remove_ideas = HSM_VER_GRO_open_court_legacy
			add_ideas = HSM_VER_GRO_managed_court_legacy
		}
	}
}
