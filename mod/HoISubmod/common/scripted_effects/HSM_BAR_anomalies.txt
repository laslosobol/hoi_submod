# Native hook provenance and complete case coverage: HSM_BAR_anomalies_coverage.txt.
HSM_BAR_anomalies_initialize = {
    if = {
        limit = { tag = BAR has_country_flag = HSM_BAR_active }
        if = {
            limit = { NOT = { has_country_flag = HSM_BAR_case_initialized } }
            set_country_flag = HSM_BAR_case_initialized
            set_global_flag = HSM_BAR_case_rock_exists
            set_global_flag = HSM_BAR_case_titan_exists
            set_global_flag = HSM_BAR_case_entity_exists
            set_global_flag = HSM_BAR_case_amulet_exists
            set_global_flag = HSM_BAR_case_chalice_exists
            set_global_flag = HSM_BAR_case_rosa_exists
            set_global_flag = HSM_BAR_case_sombra_historical
        }
        HSM_BAR_anomalies_pulse = yes
    }
}
HSM_BAR_anomalies_pulse = {
    if = {
        limit = { tag = BAR has_country_flag = HSM_BAR_active }
        # Native facts only. No known/investigated flags are granted here.
        if = {
            limit = {
                OR = {
                    has_country_flag = HSM_BAR_titan_was_dismantled
                    has_country_flag = bar_titan_tech
                    has_idea = BAR_leopold_titan_armor
                }
            }
            set_global_flag = HSM_BAR_case_titan_dismantled
        }
        if = {
            limit = {
                OR = {
                    has_country_flag = bar_elemele5
                    AND = {
                        has_variable = titanHPmax
                        has_variable = titanHP
                        check_variable = { var = titanHP value = 0 compare = less_than_or_equals }
                    }
                }
            }
            set_global_flag = HSM_BAR_case_titan_defeated
        }
        if = {
            limit = { has_country_flag = BAR_bird_saved }
            set_global_flag = HSM_BAR_case_plants_manifested
            set_global_flag = HSM_BAR_case_amulet_demonstrated
        }
        if = {
            limit = { 624 = { nuclear_reactor > 0 } }
            set_global_flag = HSM_BAR_case_reactor_built
        }
        for_each_scope_loop = {
            array = global.HSM_BAR_case_crowns
            if = {
                limit = { has_country_flag = HSM_BAR_case_crown_leader }
                if = {
                    limit = { exists = yes has_country_leader_with_trait = crown_of_mind_control }
                    HSM_BAR_case_crown_record_site = yes
                }
                else = {
                    set_country_flag = HSM_BAR_case_crown_missing
                    clr_country_flag = HSM_BAR_case_crown_leader
                }
            }
        }
    }
}
