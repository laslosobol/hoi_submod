HSM_VER_REG_maintain = {
	HSM_VER_REG_queue_estates = yes
	# A lost deadline event must not leave either House's assent permanently pending.
	if = {
		limit = { has_country_flag = HSM_VER_REG_started NOT = { has_country_flag = HSM_VER_REG_pending has_country_flag = HSM_VER_REG_regency has_country_flag = HSM_VER_REG_compact has_country_flag = HSM_VER_REG_majority has_country_flag = HSM_VER_REG_refused has_country_flag = HSM_VER_REG_majority_due } OR = { has_country_flag = HSM_VER_REG_propose_regency has_country_flag = HSM_VER_REG_propose_civil } }
		HSM_VER_REG_cancel_proposal = yes
		country_event = { id = hsm_ver_regency.15 days = 1 }
	}
	if = {
		limit = { has_country_flag = HSM_VER_REG_regency }
		if = {
			limit = { HSM_VER_REG_local_adult = yes }
			clr_country_flag = HSM_VER_REG_regency remove_ideas = HSM_VER_REG_joint_guardianship
			set_country_flag = HSM_VER_REG_majority_due
		}
		else_if = { limit = { NOT = { HSM_VER_REG_local_minor = yes } } HSM_VER_REG_end_regency = yes }
		else_if = { limit = { OR = { NOT = { has_government = neutrality } NOT = { has_country_leader = { character = AVI_lady_katrina_avian ruling_only = yes } } has_completed_focus = the_second_winterfest_ball has_completed_focus = communists_win has_completed_focus = loyalists_win has_completed_focus = fascists_win } } HSM_VER_REG_end_regency = yes }
	}
	if = { limit = { HSM_VER_REG_can_succeed = yes NOT = { has_country_flag = HSM_VER_REG_majority_pending } } set_country_flag = { flag = HSM_VER_REG_majority_pending value = 1 days = 30 } country_event = { id = hsm_ver_regency.14 days = 1 } }
	if = { limit = { has_country_flag = HSM_VER_REG_majority_due OR = { has_global_flag = GRI_grover_vi_dead NOT = { has_character = GRI_emperor_grover_vi } NOT = { has_government = neutrality } NOT = { has_country_leader = { character = AVI_lady_katrina_avian ruling_only = yes } } has_completed_focus = the_second_winterfest_ball has_completed_focus = communists_win has_completed_focus = loyalists_win has_completed_focus = fascists_win } } clr_country_flag = HSM_VER_REG_majority_due HSM_VER_REG_end_regency = yes }

	HSM_VER_REG_queue_scenes = yes
}
