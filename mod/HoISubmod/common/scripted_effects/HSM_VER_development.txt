# Only recognised lower EaW laws advance; earned and special development is kept.
# Preview only the applicable destination law. Native catch-up steps run hidden
# so tooltip expansion never lists every intermediate tier.

HSM_VER_DEV_ensure_poverty1 = {
	if = {
		limit = {
			custom_trigger_tooltip = {
				tooltip = HSM_VER_UI_DOM_below_poverty1_tt
				OR = { has_idea = poverty }
			}
		}
		effect_tooltip = { add_ideas = poverty1 }
		hidden_effect = {
			if = {
				limit = { has_idea = poverty }
				decrease_poverty = yes
			}
		}
	}
}
