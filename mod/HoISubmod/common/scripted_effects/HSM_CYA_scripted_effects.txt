HSM_CYA_install_dawnclaw_programs = {
	if = {
		limit = { NOT = { has_country_flag = HSM_CYA_dynamic_modifiers_installed } }
		set_country_flag = HSM_CYA_dynamic_modifiers_installed
		add_dynamic_modifier = { modifier = HSM_CYA_industrial_program_dynamic_modifier }
		add_dynamic_modifier = { modifier = HSM_CYA_military_program_dynamic_modifier }
		add_dynamic_modifier = { modifier = HSM_CYA_imperial_legitimacy_dynamic_modifier }
	}
}
