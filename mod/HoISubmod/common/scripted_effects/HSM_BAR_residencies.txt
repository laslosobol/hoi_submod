HSM_BAR_residencies_refresh = {
 hidden_effect = {
  HSM_BAR_residencies_clear_values = yes
  if = {
   limit = { HSM_BAR_residencies_actor = yes }
   if = { limit = { has_country_flag = HSM_BAR_residencies_equus_method } add_to_variable = { HSM_BAR_residencies_attrition = -0.01 } }
   if = {
    limit = { has_country_flag = HSM_BAR_residencies_griffonia_method }
    if = { limit = { has_dlc = "La Resistance" } add_to_variable = { HSM_BAR_residencies_cost = -0.03 } }
    else = { add_to_variable = { HSM_BAR_residencies_recon = 0.005 } }
   }
   if = { limit = { has_country_flag = HSM_BAR_residencies_north_method } add_to_variable = { HSM_BAR_residencies_recon = 0.01 } }
   if = { limit = { has_country_flag = HSM_BAR_residencies_riverlands_method } add_to_variable = { HSM_BAR_residencies_dig_in = 0.05 } }
   if = {
    limit = { has_country_flag = HSM_BAR_residencies_comparison_method }
    if = { limit = { has_country_flag = HSM_BAR_residencies_zebrica_listeners } add_to_variable = { HSM_BAR_residencies_supply = -0.02 } }
    else = {
     if = { limit = { has_dlc = "La Resistance" } add_to_variable = { HSM_BAR_residencies_outcome = 0.03 } }
     else = { add_to_variable = { HSM_BAR_residencies_recon = 0.01 } }
    }
   }
   if = {
    limit = { has_country_flag = HSM_BAR_residencies_aftermath_done }
    if = {
     limit = { has_country_flag = HSM_BAR_residencies_watch }
     add_to_variable = { HSM_BAR_residencies_upkeep = -0.03 }
     if = { limit = { has_dlc = "La Resistance" } add_to_variable = { HSM_BAR_residencies_network = 0.05 } add_to_variable = { HSM_BAR_residencies_outcome = 0.05 } }
     else = { add_to_variable = { HSM_BAR_residencies_recon = 0.02 } }
    }
    else = {
     add_to_variable = { HSM_BAR_residencies_supply = -0.01 }
     if = { limit = { has_dlc = "La Resistance" } add_to_variable = { HSM_BAR_residencies_defence = 1 } }
     else = { add_to_variable = { HSM_BAR_residencies_recon = 0.01 } }
    }
   }
   else_if = {
    limit = { has_country_flag = HSM_BAR_residencies_contact_done }
    if = {
     limit = { has_country_flag = HSM_BAR_residencies_contact_follow }
     if = {
      limit = { has_dlc = "La Resistance" }
      if = { limit = { has_country_flag = HSM_BAR_cult_channel_retained } add_to_variable = { HSM_BAR_residencies_network = 0.05 } add_to_variable = { HSM_BAR_residencies_capture = 0.05 } }
      else = { add_to_variable = { HSM_BAR_residencies_network = 0.025 } }
     }
     else = { add_to_variable = { HSM_BAR_residencies_recon = 0.01 } }
    }
    else = {
     if = { limit = { has_dlc = "La Resistance" } add_to_variable = { HSM_BAR_residencies_defence = 1 } }
     else = { add_to_variable = { HSM_BAR_residencies_recon = 0.005 } }
    }
    if = {
     limit = { has_country_flag = HSM_BAR_residencies_exchange_done }
     if = {
      limit = { has_country_flag = HSM_BAR_residencies_exchange_follow }
      add_to_variable = { HSM_BAR_residencies_upkeep = -0.02 }
      if = { limit = { has_dlc = "La Resistance" } add_to_variable = { HSM_BAR_residencies_outcome = 0.05 } }
      else = { add_to_variable = { HSM_BAR_residencies_recon = 0.005 } }
     }
     else = { add_to_variable = { HSM_BAR_residencies_supply = -0.01 } }
    }
   }
   if = {
    limit = {
     OR = {
      has_country_flag = HSM_BAR_residencies_equus_method has_country_flag = HSM_BAR_residencies_griffonia_method
      has_country_flag = HSM_BAR_residencies_north_method has_country_flag = HSM_BAR_residencies_riverlands_method
      has_country_flag = HSM_BAR_residencies_comparison_method has_country_flag = HSM_BAR_residencies_contact_done
     }
    }
    if = { limit = { NOT = { has_dynamic_modifier = HSM_BAR_residencies_methods } } add_dynamic_modifier = { modifier = HSM_BAR_residencies_methods } }
    force_update_dynamic_modifier = yes
   }
  }
  else = { HSM_BAR_residencies_teardown = yes }
 }
}
