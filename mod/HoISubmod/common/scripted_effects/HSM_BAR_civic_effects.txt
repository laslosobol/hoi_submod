HSM_BAR_civic_refresh = {
 hidden_effect = {
  HSM_BAR_civic_clear_values = yes
  if = {
   limit = { HSM_BAR_civic_available = yes has_country_flag = HSM_BAR_civic_republic_awarded }
   if = {
    limit = { has_country_flag = HSM_BAR_civic_registry_awarded }
    add_to_variable = { HSM_BAR_civic_mobilization = 0.1 }
   }
   if = {
    limit = { has_country_flag = HSM_BAR_civic_civil_service_awarded }
    add_to_variable = { HSM_BAR_civic_pp = 0.08 }
   }
   if = {
    limit = { has_country_flag = HSM_BAR_civic_accounts_awarded }
    add_to_variable = { HSM_BAR_civic_pp = 0.05 }
   }
   if = {
    limit = { has_country_flag = HSM_BAR_civic_cabinet_awarded }
    add_to_variable = { HSM_BAR_civic_pp = 0.05 }
   }
   if = {
    limit = { has_country_flag = HSM_BAR_civic_foreign_office_awarded }
    add_to_variable = { HSM_BAR_civic_relations = -0.1 }
   }
   if = {
    limit = { has_country_flag = HSM_BAR_civic_appeals_awarded }
    # Native resistance_target is a fraction: -0.02 displays as -2%, not -200%.
    add_to_variable = { HSM_BAR_civic_resistance = -0.02 }
    add_to_variable = { HSM_BAR_civic_stability = 0.03 }
   }
   if = {
    limit = { has_country_flag = HSM_BAR_civic_local_works_awarded }
    add_to_variable = { HSM_BAR_civic_infrastructure = 0.1 }
   }
   if = {
    limit = { has_country_flag = HSM_BAR_civic_inspectors_awarded }
    add_to_variable = { HSM_BAR_civic_pp = 0.05 }
   }
   if = {
    limit = { has_country_flag = HSM_BAR_civic_equal_services_awarded }
    add_to_variable = { HSM_BAR_civic_compliance = 0.02 }
   }
   if = {
    limit = { has_country_flag = HSM_BAR_civic_ordinary_state_awarded }
    add_to_variable = { HSM_BAR_civic_stability = 0.05 }
    add_to_variable = { HSM_BAR_civic_pp = 0.05 }
   }
   if = {
    limit = { has_country_flag = HSM_BAR_civic_local_mandates has_country_flag = HSM_BAR_civic_municipalities_settled }
    add_to_variable = { HSM_BAR_civic_construction = 0.05 }
    add_to_variable = { HSM_BAR_civic_pp = -0.03 }
   }
   else_if = {
    limit = { has_country_flag = HSM_BAR_civic_district_mandates has_country_flag = HSM_BAR_civic_municipalities_settled }
    add_to_variable = { HSM_BAR_civic_pp = 0.05 }
    add_to_variable = { HSM_BAR_civic_stability = -0.02 }
   }
   if = {
    limit = { has_country_flag = HSM_BAR_civic_circuit_courts has_country_flag = HSM_BAR_civic_courts_settled }
    add_to_variable = { HSM_BAR_civic_resistance = -0.05 }
    add_to_variable = { HSM_BAR_civic_pp = -0.02 }
   }
   else_if = {
    limit = { has_country_flag = HSM_BAR_civic_permanent_bench has_country_flag = HSM_BAR_civic_courts_settled }
    add_to_variable = { HSM_BAR_civic_stability = 0.04 }
    add_to_variable = { HSM_BAR_civic_compliance = -0.02 }
   }
   if = {
    limit = { has_country_flag = HSM_BAR_civic_local_reserve has_country_flag = HSM_BAR_civic_defence_settled }
    add_to_variable = { HSM_BAR_civic_core_defence = 0.10 }
    add_to_variable = { HSM_BAR_civic_training = 0.03 }
   }
   else_if = {
    limit = { has_country_flag = HSM_BAR_civic_central_depots has_country_flag = HSM_BAR_civic_defence_settled }
    add_to_variable = { HSM_BAR_civic_supply = -0.08 }
    add_to_variable = { HSM_BAR_civic_recovery = 0.05 }
    add_to_variable = { HSM_BAR_civic_pp = -0.03 }
   }
   if = {
    limit = { has_country_flag = HSM_BAR_civic_open_tenders has_country_flag = HSM_BAR_civic_procurement_settled }
    add_to_variable = { HSM_BAR_civic_construction = 0.07 }
    add_to_variable = { HSM_BAR_civic_efficiency_gain = -0.03 }
   }
   else_if = {
    limit = { has_country_flag = HSM_BAR_civic_framework_orders has_country_flag = HSM_BAR_civic_procurement_settled }
    add_to_variable = { HSM_BAR_civic_output = 0.08 }
    add_to_variable = { HSM_BAR_civic_supply = -0.05 }
    add_to_variable = { HSM_BAR_civic_max_efficiency = -0.02 }
   }
   if = {
    limit = { has_country_flag = HSM_BAR_civic_local_statutes has_country_flag = HSM_BAR_civic_territorial_representation_settled }
    add_to_variable = { HSM_BAR_civic_compliance = 0.04 }
    add_to_variable = { HSM_BAR_civic_pp = -0.03 }
   }
   else_if = {
    limit = { has_country_flag = HSM_BAR_civic_equal_franchise has_country_flag = HSM_BAR_civic_territorial_representation_settled }
    add_to_variable = { HSM_BAR_civic_stability = 0.03 }
    add_to_variable = { HSM_BAR_civic_resistance = -0.03 }
    add_to_variable = { HSM_BAR_civic_output = -0.01 }
   }
   if = {
    limit = { NOT = { has_dynamic_modifier = HSM_BAR_civic_government } }
    add_dynamic_modifier = { modifier = HSM_BAR_civic_government }
   }
   force_update_dynamic_modifier = yes
  }
  else = { HSM_BAR_civic_teardown = yes }
 }
}
# Events are immediate. A suspended event is reissued after access returns.
HSM_BAR_civic_pulse = {
 HSM_BAR_civic_refresh = yes
 if = {
  limit = { HSM_BAR_civic_available = yes }
  if = {
   limit = {
    has_country_flag = HSM_BAR_civic_municipalities_awarded
    NOT = { has_country_flag = HSM_BAR_civic_municipalities_settled }
    NOT = { has_country_flag = HSM_BAR_civic_municipalities_pending }
   }
   set_country_flag = HSM_BAR_civic_municipalities_pending
   country_event = { id = HSM_BAR_civic.1 }
  }
  if = {
   limit = {
    has_country_flag = HSM_BAR_civic_courts_awarded
    NOT = { has_country_flag = HSM_BAR_civic_courts_settled }
    NOT = { has_country_flag = HSM_BAR_civic_courts_pending }
   }
   set_country_flag = HSM_BAR_civic_courts_pending
   country_event = { id = HSM_BAR_civic.2 }
  }
  if = {
   limit = {
    has_country_flag = HSM_BAR_civic_defence_awarded
    NOT = { has_country_flag = HSM_BAR_civic_defence_settled }
    NOT = { has_country_flag = HSM_BAR_civic_defence_pending }
   }
   set_country_flag = HSM_BAR_civic_defence_pending
   country_event = { id = HSM_BAR_civic.3 }
  }
  if = {
   limit = {
    has_country_flag = HSM_BAR_civic_procurement_awarded
    NOT = { has_country_flag = HSM_BAR_civic_procurement_settled }
    NOT = { has_country_flag = HSM_BAR_civic_procurement_pending }
   }
   set_country_flag = HSM_BAR_civic_procurement_pending
   country_event = { id = HSM_BAR_civic.4 }
  }
  if = {
   limit = {
    has_country_flag = HSM_BAR_civic_territorial_representation_awarded
    NOT = { has_country_flag = HSM_BAR_civic_territorial_representation_settled }
    NOT = { has_country_flag = HSM_BAR_civic_territorial_representation_pending }
   }
   set_country_flag = HSM_BAR_civic_territorial_representation_pending
   country_event = { id = HSM_BAR_civic.5 }
  }
 }
}
