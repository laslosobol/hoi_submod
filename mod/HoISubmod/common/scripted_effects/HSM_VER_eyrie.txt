HSM_VER_eyr_start_campaign = {
	hidden_effect = { HSM_VER_enable_negotiated_campaign = yes }
	if = {
		limit = { hidden_trigger = { NOT = { has_country_flag = HSM_EYR_campaign_active } } }
		hidden_effect = { set_country_flag = HSM_EYR_campaign_active }
		country_event = { id = hsm_ver_eyrie.1 days = 1 }
	}
}
# The existing ERI register serialises both Houses' proposals. Response events differ.
HSM_VER_eyr_offer_armistice = {
	if = {
		limit = { HSM_VER_can_offer_armistice = yes }
		log = "[GetDateText]: Verenia: armistice offer sent to Erie."
		ERI = {
			set_country_flag = HSM_VER_negotiation_revision_2
			set_country_flag = HSM_VER_armistice_from_eyrie
			set_country_flag = HSM_VER_armistice_pending
			set_country_flag = { flag = HSM_VER_armistice_reply_window value = 1 days = 29 }
			set_country_flag = { flag = HSM_VER_armistice_retry value = 1 days = 90 }
			clr_country_flag = HSM_VER_armistice_refused
			country_event = { id = hsm_ver_eyrie.110 days = 1 }
		}
		EYR = { country_event = { id = hsm_ver_eyrie.119 days = 30 } }
	}
}

HSM_VER_eyr_offer_charter = {
	if = {
		limit = {
			HSM_VER_diplomatic_routes_open = yes HSM_VER_negotiated_peace_valid = yes HSM_VER_eyr_government_available = yes
			ERI = { NOT = { has_country_flag = HSM_VER_charter_signed has_country_flag = HSM_VER_charter_pending has_country_flag = HSM_VER_charter_retry } }
		}
		ERI = {
			set_country_flag = HSM_VER_charter_from_eyrie
			set_country_flag = HSM_VER_charter_pending
			set_country_flag = { flag = HSM_VER_charter_retry value = 1 days = 120 }
			clr_country_flag = HSM_VER_charter_refused
			clr_country_flag = HSM_VER_ducal_presidency
			if = { limit = { is_ai = yes } HSM_VER_ratify_charter = yes EYR = { country_event = { id = hsm_ver_eyrie.132 days = 1 } } }
			else = { country_event = { id = hsm_ver_eyrie.130 days = 1 } }
		}
		EYR = { country_event = { id = hsm_ver_eyrie.139 days = 45 } }
	}
}

			}
		}
		else = { add_tech_bonus = { name = HSM_EYR_reconstruction_bonus bonus = 1 uses = 1 category = construction_tech } }
	}
}
