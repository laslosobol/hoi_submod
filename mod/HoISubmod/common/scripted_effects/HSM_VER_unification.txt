HSM_VER_UNI_start_convention = {

		limit = { HSM_VER_UNI_can_start = yes }
		set_country_flag = HSM_VER_UNI_started
		set_country_flag = HSM_VER_UNI_leader
		if = { limit = { HSM_VER_UNI_diplomatic_context = yes } set_country_flag = HSM_VER_UNI_diplomatic }
		else = {
			set_country_flag = HSM_VER_UNI_military
			if = {
				limit = { OR = { AND = { tag = ERI EYR = { exists = no } } AND = { tag = EYR ERI = { exists = no } } } }
				set_country_flag = HSM_VER_UNI_preannexed_rival
			}
		}
		if = {
			limit = { tag = ERI }
			set_country_flag = HSM_VER_UNI_president_ERI
			EYR = {
				if = {
					limit = { HSM_VER_UNI_candidate_for_ERI = yes NOT = { has_country_flag = HSM_VER_UNI_invited_ERI } }
					set_country_flag = HSM_VER_UNI_invited_ERI
					set_country_flag = HSM_VER_UNI_invite_pending_ERI

					country_event = { id = hsm_ver_uni.11 days = 1 }
				}
			}
			AVI = {
				if = {
					limit = { HSM_VER_UNI_candidate_for_EYR = yes NOT = { has_country_flag = HSM_VER_UNI_invited_EYR } }
					set_country_flag = HSM_VER_UNI_invited_EYR
					set_country_flag = HSM_VER_UNI_invite_pending_EYR
					set_country_flag = { flag = HSM_VER_UNI_reply_wait_EYR value = 1 days = 45 }
					country_event = { id = hsm_ver_uni.11 days = 1 }
				}
			}
		}
		if = {
			limit = { tag = AVI }
			set_country_flag = HSM_VER_UNI_president_AVI

# A House receives its physical works only once, including across failed presidencies.
HSM_VER_UNI_grant_works = {
	if = {
		limit = { NOT = { has_country_flag = HSM_VER_UNI_works_received } }
		set_country_flag = HSM_VER_UNI_works_received
		if = {
			limit = { tag = ERI owns_state = 401 }
			401 = {
				add_extra_state_shared_building_slots = 1
				if = {
					limit = { free_building_slots = { building = industrial_complex size > 0 include_locked = yes } }
					add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
				}
			}
		}
		if = {
			limit = { tag = EYR owns_state = 400 }
			400 = {
				add_extra_state_shared_building_slots = 1
				if = {
					limit = { free_building_slots = { building = industrial_complex size > 0 include_locked = yes } }
					add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
				}
			}
		}
		if = {
			limit = { tag = AVI owns_state = 470 }
			470 = {
				add_extra_state_shared_building_slots = 1
				if = {
					limit = { free_building_slots = { building = industrial_complex size > 0 include_locked = yes } }
					add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
				}
			}
		}
	}
}

HSM_VER_UNI_found_union = {
	# Validate the entire compact before the first annexation invalidates the old charter.
	if = {
		limit = { HSM_VER_UNI_can_found = yes }
		set_country_flag = HSM_VER_UNI_committing
		if = {
			limit = { tag = ERI }
			if = {
				limit = { always = yes }
				# Transfer actual custody before the holder disappears; never recruit a child.
				if = {
					limit = {
						EYR = { has_character = GRI_emperor_grover_vi }
						NOT = { has_global_flag = GRI_grover_vi_dead any_country = { NOT = { tag = EYR } has_character = GRI_emperor_grover_vi } }
					}
					EYR = { GRI_emperor_grover_vi = { set_nationality = ERI } }
				}
				if = { limit = { EYR = { exists = yes } } annex_country = { target = EYR transfer_troops = yes } }
				# An absent rival reached here only through recorded military victory.
				if = {
					limit = { EYR = { exists = no } owns_state = 391 391 = { is_fully_controlled_by = ERI } owns_state = 400 400 = { is_fully_controlled_by = ERI } }
					set_country_flag = HSM_VER_UNI_absorbed_EYR
					391 = { add_core_of = ERI }
					400 = { add_core_of = ERI }
				}
			}
			if = {
				limit = { AVI = { has_country_flag = HSM_VER_UNI_member_ERI has_country_flag = HSM_VER_UNI_consent_ERI } }
				# Transfer actual custody before the holder disappears; never recruit a child.
				if = {
					limit = {
						AVI = { has_character = GRI_emperor_grover_vi }
						NOT = { has_global_flag = GRI_grover_vi_dead any_country = { NOT = { tag = AVI } has_character = GRI_emperor_grover_vi } }
					}
					AVI = { GRI_emperor_grover_vi = { set_nationality = ERI } }
				}
				if = { limit = { AVI = { exists = yes } } HSM_VER_SP_inherit_avian = yes annex_country = { target = AVI transfer_troops = yes } }
				# An absent rival reached here only through recorded military victory.
				if = {
					limit = { AVI = { exists = no } owns_state = 470 470 = { is_fully_controlled_by = ERI } }
					set_country_flag = HSM_VER_UNI_absorbed_AVI
					470 = { add_core_of = ERI }
				}
			}
		}
		if = {
			limit = { tag = EYR }
			if = {
				limit = { always = yes }
				# Transfer actual custody before the holder disappears; never recruit a child.
				if = {
					limit = {
						ERI = { has_character = GRI_emperor_grover_vi }
						NOT = { has_global_flag = GRI_grover_vi_dead any_country = { NOT = { tag = ERI } has_character = GRI_emperor_grover_vi } }
					}
					ERI = { GRI_emperor_grover_vi = { set_nationality = EYR } }
				}
				if = { limit = { ERI = { exists = yes } } annex_country = { target = ERI transfer_troops = yes } }
				# An absent rival reached here only through recorded military victory.
				if = {
					limit = { ERI = { exists = no } owns_state = 401 401 = { is_fully_controlled_by = EYR } owns_state = 500 500 = { is_fully_controlled_by = EYR } owns_state = 673 673 = { is_fully_controlled_by = EYR } }
					set_country_flag = HSM_VER_UNI_absorbed_ERI
					401 = { add_core_of = EYR }
					500 = { add_core_of = EYR }
					673 = { add_core_of = EYR }
				}
			}
			if = {
				limit = { AVI = { has_country_flag = HSM_VER_UNI_member_EYR has_country_flag = HSM_VER_UNI_consent_EYR } }
				# Transfer actual custody before the holder disappears; never recruit a child.
				if = {

HSM_VER_UNI_maintain_campaign = {
	HSM_VER_UNI_migrate_cosmetic = yes
	HSM_VER_SP_maintain = yes
	if = {
		limit = { has_country_flag = HSM_VER_UNI_leader NOT = { has_country_flag = HSM_VER_UNI_founded has_country_flag = HSM_VER_UNI_committing } }
		# Only political dissolution ends a mandate. War and occupation have no expiry.
		if = { limit = { NOT = { HSM_VER_UNI_mandate_survives = yes } } HSM_VER_UNI_close_convention = yes }
		else_if = {
			limit = { NOT = { HSM_VER_UNI_context = yes } }
			HSM_VER_UNI_suspend_convention = yes
		}
		else_if = {
			limit = { has_country_flag = HSM_VER_UNI_suspended }
			clr_country_flag = HSM_VER_UNI_suspended
			if = { limit = { has_country_flag = HSM_VER_UNI_customs NOT = { has_country_flag = HSM_VER_UNI_administration } } set_country_flag = { flag = HSM_VER_UNI_customs_wait value = 1 days = 60 } }

		}
		HSM_VER_UNI_resume_invitations = yes
	}
	# Completed focuses can finish a deferred reward after a temporary suspension.
	if = { limit = { has_completed_focus = HSM_VER_UNI_convention NOT = { has_country_flag = HSM_VER_COOP_shared_history } } HSM_VER_UNI_start_convention = yes }
	if = { limit = { has_completed_focus = HSM_VER_UNI_customs_union } HSM_VER_UNI_establish_customs = yes }
	if = { limit = { has_completed_focus = HSM_VER_UNI_common_works } HSM_VER_UNI_build_common_works = yes }
	if = { limit = { has_completed_focus = HSM_VER_UNI_common_administration } HSM_VER_UNI_establish_administration = yes }
	if = { limit = { has_completed_focus = HSM_VER_UNI_unified_command } HSM_VER_UNI_establish_command = yes }

	HSM_VER_UNI_dispatch_ratification = yes
	if = { limit = { has_completed_focus = HSM_VER_UNI_constitutional_act } HSM_VER_UNI_found_union = yes }
	if = { limit = { has_completed_focus = HSM_VER_UNI_settled_institutions } HSM_VER_UNI_settle_institutions = yes }
