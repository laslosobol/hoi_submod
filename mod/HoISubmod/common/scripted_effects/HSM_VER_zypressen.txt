HSM_ZYP_queue_coronation = {
 if = {
  limit = { hidden_trigger = { HSM_ZYP_entry = yes has_completed_focus = HSM_ZYP_imperial_coronation NOT = { has_country_flag = HSM_ZYP_crowned has_country_flag = HSM_ZYP_coronation_pending has_country_flag = HSM_ZYP_coronation_retry } } }
  hidden_effect = { set_country_flag = { flag = HSM_ZYP_coronation_pending value = 1 days = 30 } }
  country_event = { id = hsm_ver_zypressen.1 days = 1 }
 }
}
HSM_ZYP_crown_emperor = {

  limit = { HSM_ZYP_entry = yes has_country_flag = HSM_ZYP_coronation_pending NOT = { has_country_flag = HSM_ZYP_crowned } }
  set_country_flag = HSM_ZYP_crowned clr_country_flag = HSM_ZYP_coronation_pending
  add_stability = 0.07 add_ideas = HSM_ZYP_imperial_chancery
 }
}

   country_event = { id = hsm_ver_zypressen.10 days = 1 }
  }
 }
}
# Recipient scope. Agreements never change autonomy, ownership, cores or rulers.
HSM_ZYP_sign_autonomous = {
 hidden_effect = {
 set_country_flag = HSM_ZYP_signed set_country_flag = HSM_ZYP_autonomous
 clr_country_flag = HSM_ZYP_offer_pending clr_country_flag = HSM_ZYP_counter_pending
 clr_country_flag = HSM_ZYP_refused
 }
 HSM_ZYP_refresh_agreement = yes
}
HSM_ZYP_refresh_agreement = {
 if = {
  limit = { hidden_trigger = { HSM_ZYP_agreement_valid = yes } }
  hidden_effect = {
  if = {
   limit = { has_country_flag = HSM_ZYP_autonomous }
   if = { limit = { NOT = { has_idea = HSM_ZYP_autonomous_crown } } add_ideas = HSM_ZYP_autonomous_crown }
  }
  else = { if = { limit = { NOT = { has_idea = HSM_ZYP_common_ministries } } add_ideas = HSM_ZYP_common_ministries } }
  }
  if = {
   limit = { hidden_trigger = { NOT = { has_country_flag = HSM_ZYP_workshop_grant } } }
   hidden_effect = { set_country_flag = HSM_ZYP_workshop_grant }
   add_tech_bonus = { name = HSM_ZYP_industry_bonus bonus = 1 uses = 1 category = industry }
  }
  if = {
   limit = { hidden_trigger = { AVI = { has_completed_focus = HSM_ZYP_common_colleges } NOT = { has_country_flag = HSM_ZYP_college_grant } } }
   hidden_effect = { set_country_flag = HSM_ZYP_college_grant }
   add_tech_bonus = { name = HSM_ZYP_electronics_bonus bonus = 1 uses = 1 category = electronics }
  }
 }
 else = { hidden_effect = { remove_ideas = HSM_ZYP_autonomous_crown remove_ideas = HSM_ZYP_common_ministries } }
}
# A physical reward only on owned, fully controlled land, with research if no capacity remains.
HSM_ZYP_civilian_works = {
 if = {
  limit = { hidden_trigger = { any_owned_state = { is_fully_controlled_by = ROOT free_building_slots = { building = industrial_complex size > 0 include_locked = yes } } } }
  random_owned_state = {
   limit = { hidden_trigger = { is_fully_controlled_by = ROOT free_building_slots = { building = industrial_complex size > 0 include_locked = yes } } }
   add_extra_state_shared_building_slots = 1
   add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
  }
 }
 else = { add_tech_bonus = { name = HSM_ZYP_construction_bonus bonus = 1 uses = 1 category = construction_tech } }
}
HSM_ZYP_military_works = {
 if = {
  limit = { hidden_trigger = { any_owned_state = { is_fully_controlled_by = ROOT free_building_slots = { building = arms_factory size > 0 include_locked = yes } } } }
  random_owned_state = {
   limit = { hidden_trigger = { is_fully_controlled_by = ROOT free_building_slots = { building = arms_factory size > 0 include_locked = yes } } }
   add_extra_state_shared_building_slots = 1
   add_building_construction = { type = arms_factory level = 1 instant_build = yes }
  }
 }
 else = { add_tech_bonus = { name = HSM_ZYP_industry_bonus bonus = 1 uses = 1 category = industry } }
}
HSM_ZYP_goals_west = {
 every_country = {
  limit = { HSM_ZYP_west_opponent = yes }
  AVI = {
   if = {
    limit = { NOT = { has_wargoal_against = PREV } }
    if = {
     limit = { PREV = { HSM_ZYP_west_owner = yes } }
     create_wargoal = { type = take_state_focus target = PREV generator = { 390 208 436 457 380 396 379 383 376 517 421 472 435 524 502 523 516 407 473 377 406 408 } }
    }
    else = {
     # A foreign overlord may own no regional state. This grants a goal, never annexation.
     create_wargoal = { type = annex_everything target = PREV }
    }
   }
  }
 }
}

# Only the first unresolved, unlocked theatre is reviewed; never declare war.
HSM_ZYP_review_frontiers = {
 if = {
  limit = { HSM_ZYP_active = yes has_war = no }
  if = { limit = { has_completed_focus = HSM_ZYP_western_campaign NOT = { HSM_ZYP_west_settled = yes } } HSM_ZYP_goals_west = yes }
  else_if = { limit = { has_completed_focus = HSM_ZYP_northern_campaign NOT = { HSM_ZYP_north_settled = yes } } HSM_ZYP_goals_north = yes }
  else_if = { limit = { has_completed_focus = HSM_ZYP_southern_campaign NOT = { HSM_ZYP_south_settled = yes } } HSM_ZYP_goals_south = yes }
  else_if = { limit = { has_completed_focus = HSM_ZYP_eastern_campaign NOT = { HSM_ZYP_east_settled = yes } } HSM_ZYP_goals_east = yes }
 }
}
HSM_ZYP_maintain = {
 every_country = {
  limit = { OR = { has_country_flag = HSM_ZYP_signed has_country_flag = HSM_ZYP_offer_pending has_country_flag = HSM_ZYP_counter_pending } }
  if = {
   limit = { OR = { NOT = { HSM_ZYP_dynastic_crown = yes } AVI = { NOT = { HSM_ZYP_government = yes } } has_war_with = AVI } }
   clr_country_flag = HSM_ZYP_signed clr_country_flag = HSM_ZYP_autonomous
   clr_country_flag = HSM_ZYP_offer_pending clr_country_flag = HSM_ZYP_counter_pending
  }
  HSM_ZYP_refresh_agreement = yes
 }
 if = { limit = { HSM_ZYP_active = yes } if = { limit = { NOT = { has_idea = HSM_ZYP_imperial_chancery } } add_ideas = HSM_ZYP_imperial_chancery } }
 else = { remove_ideas = HSM_ZYP_imperial_chancery }
 if = { limit = { HSM_ZYP_active = yes any_country = { HSM_ZYP_agreement_valid = yes } } if = { limit = { NOT = { has_idea = HSM_ZYP_dynastic_exchange } } add_ideas = HSM_ZYP_dynastic_exchange } }
 else = { remove_ideas = HSM_ZYP_dynastic_exchange }
 if = {
  limit = { HSM_ZYP_active = yes }
  if = {
   limit = { has_completed_focus = HSM_ZYP_rolls_of_the_crowns NOT = { has_country_flag = HSM_ZYP_policy_chosen has_country_flag = HSM_ZYP_policy_pending } }
   set_country_flag = { flag = HSM_ZYP_policy_pending value = 1 days = 30 }
   country_event = { id = hsm_ver_zypressen.2 days = 1 }
  }
  if = {
   limit = { has_completed_focus = HSM_ZYP_repair_the_arteries NOT = { has_country_flag = HSM_ZYP_works_chosen has_country_flag = HSM_ZYP_works_pending } }
   set_country_flag = { flag = HSM_ZYP_works_pending value = 1 days = 30 }
   country_event = { id = hsm_ver_zypressen.4 days = 1 }
  }
  if = {
   limit = { has_completed_focus = HSM_ZYP_the_muster_dispute NOT = { has_country_flag = HSM_ZYP_muster_chosen has_country_flag = HSM_ZYP_muster_pending } }
   set_country_flag = { flag = HSM_ZYP_muster_pending value = 1 days = 30 }
   country_event = { id = hsm_ver_zypressen.5 days = 1 }
  }
 }
}
