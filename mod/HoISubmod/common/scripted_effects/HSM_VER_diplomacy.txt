HSM_VER_enable_negotiated_campaign = {
	if = {
		limit = { ERI = { NOT = { has_country_flag = HSM_VER_campaign_active } } }
		ERI = { set_country_flag = HSM_VER_campaign_active }

			limit = { AVI = { has_completed_focus = treaty_of_friendship } HSM_VER_houses_at_peace = yes ERI = { NOT = { has_country_flag = HSM_VER_treaty_broken } } }
			HSM_VER_sign_treaty = yes
		}
	}
}

HSM_VER_sign_armistice = {

			ERI = { set_country_flag = HSM_VER_joint_inspection }
		}
		if = { limit = { ERI = { has_border_war_with = EYR } } cancel_border_war = { attacker = 401 defender = 400 } cancel_border_war = { attacker = 400 defender = 401 } }

			clr_country_flag = HSM_VER_armistice_pending
			clr_country_flag = HSM_VER_armistice_offer_unresolved

			else = { set_country_flag = { flag = HSM_VER_armistice value = 1 days = 365 } }
			set_truce = { target = EYR days = 365 }
			remove_wargoal = { type = annex_everything target = EYR }
			remove_wargoal = { type = puppet_wargoal_focus target = EYR }
			country_event = { id = hsm_verenia.22 days = 1 }
		}
		EYR = {
			if = { limit = { has_country_flag = HSM_VER_diplomatic_route } set_country_flag = HSM_VER_armistice }
			else = { set_country_flag = { flag = HSM_VER_armistice value = 1 days = 365 } }

			limit = { ERI = { has_country_flag = HSM_VER_joint_inspection } }
			ERI = { add_ideas = HSM_VER_frontier_inspections }

# ERI's register retains unresolved offers after their timed pending flags expire.
# Legacy and Eyrie-produced untimed offers close only after the original cooldown,
# which outlasts every queued reply and timeout. Recovery never restarts that clock.
HSM_VER_reconcile_negotiation_timeouts = {
	if = {
		limit = {
			OR = {
				AND = { has_country_flag = HSM_VER_armistice_pending NOT = { has_country_flag = HSM_VER_armistice_retry } }
				AND = { has_country_flag = HSM_VER_armistice_offer_unresolved NOT = { has_country_flag = HSM_VER_armistice_pending } }
			}
		}
		clr_country_flag = HSM_VER_armistice_pending
		clr_country_flag = HSM_VER_armistice_offer_unresolved
		clr_country_flag = HSM_VER_armistice_from_eyrie
		if = { limit = { NOT = { has_country_flag = HSM_VER_armistice has_country_flag = HSM_VER_treaty_signed } } set_country_flag = HSM_VER_armistice_refused }
	}
	if = {
		limit = {
			OR = {
				AND = { has_country_flag = HSM_VER_treaty_pending NOT = { has_country_flag = HSM_VER_treaty_retry } }
				AND = { has_country_flag = HSM_VER_treaty_offer_unresolved NOT = { has_country_flag = HSM_VER_treaty_pending } }
			}
		}
		clr_country_flag = HSM_VER_treaty_pending
		clr_country_flag = HSM_VER_treaty_offer_unresolved
		clr_country_flag = HSM_VER_treaty_from_eyrie
		if = { limit = { NOT = { has_country_flag = HSM_VER_treaty_signed } } set_country_flag = HSM_VER_treaty_refused }
	}
	if = {
		limit = {
			OR = {
				AND = { has_country_flag = HSM_VER_charter_pending NOT = { has_country_flag = HSM_VER_charter_retry } }
				AND = { has_country_flag = HSM_VER_charter_offer_unresolved NOT = { has_country_flag = HSM_VER_charter_pending } }
			}
		}
		clr_country_flag = HSM_VER_charter_pending
		clr_country_flag = HSM_VER_charter_offer_unresolved
		clr_country_flag = HSM_VER_charter_from_eyrie
		if = { limit = { NOT = { has_country_flag = HSM_VER_charter_signed } } set_country_flag = HSM_VER_charter_refused }
	}
	if = {
		limit = {
			OR = {
				AND = { has_country_flag = HSM_VER_avian_pending NOT = { has_country_flag = HSM_VER_avian_retry } }
				AND = { has_country_flag = HSM_VER_avian_offer_unresolved NOT = { has_country_flag = HSM_VER_avian_pending } }
			}
		}
		clr_country_flag = HSM_VER_avian_pending
		clr_country_flag = HSM_VER_avian_offer_unresolved
		if = { limit = { AVI = { NOT = { has_country_flag = HSM_VER_congress_observer has_country_flag = HSM_VER_charter_signed has_war_with = ERI has_war_with = EYR } } } set_country_flag = HSM_VER_avian_declined }
	}

# Called daily for just the three tags. Historical signatures survive occupation;
# benefits do not. Actual inter-House war voids the agreement permanently.
HSM_VER_maintain_agreements = {
	if = { limit = { OR = { tag = ERI tag = EYR } } HSM_VER_migrate_routes = yes }
	if = {
		limit = {
			OR = { tag = ERI tag = EYR }
			HSM_VER_armistice_valid = yes ERI = { has_country_flag = HSM_VER_joint_inspection }
			NOT = { HSM_VER_negotiated_peace_valid = yes has_idea = HSM_VER_frontier_inspections }
		}
		add_ideas = HSM_VER_frontier_inspections
	}
	if = {
		limit = { has_idea = HSM_VER_frontier_inspections OR = { NOT = { HSM_VER_armistice_valid = yes } HSM_VER_negotiated_peace_valid = yes } }
		remove_ideas = HSM_VER_frontier_inspections
	}
	if = { limit = { tag = ERI } HSM_VER_reconcile_defensive_league = yes HSM_VER_reconcile_charter_suspension = yes }
	if = {
		limit = { tag = ERI has_war_with = EYR OR = { has_country_flag = HSM_VER_armistice has_country_flag = HSM_VER_treaty_signed has_country_flag = HSM_VER_armistice_pending has_country_flag = HSM_VER_treaty_pending has_country_flag = HSM_VER_charter_pending } }
		HSM_VER_void_bilateral_treaty = yes
	}
	if = {
		limit = { tag = AVI OR = { has_country_flag = HSM_VER_charter_signed has_country_flag = HSM_VER_congress_observer has_country_flag = HSM_VER_avian_charter_pending } OR = { has_war_with = ERI has_war_with = EYR } }
		clr_country_flag = HSM_VER_charter_signed
		clr_country_flag = HSM_VER_congress_observer
		clr_country_flag = HSM_VER_avian_charter_pending
		clr_country_flag = HSM_VER_avian_defence_pending
		ERI = { clr_country_flag = HSM_VER_avian_pending clr_country_flag = HSM_VER_avian_offer_unresolved }

	}
	if = {
		limit = { has_country_flag = HSM_VER_treaty_signed NOT = { HSM_VER_negotiated_peace_valid = yes } }
		remove_ideas = HSM_VER_peace_dividend
	}
	if = {
		limit = { has_country_flag = HSM_VER_treaty_signed HSM_VER_negotiated_peace_valid = yes NOT = { has_idea = HSM_VER_peace_dividend } }
		add_ideas = HSM_VER_peace_dividend
	}
	if = {
		limit = { has_country_flag = HSM_VER_charter_signed }
		if = {
			limit = { HSM_VER_charter_valid = yes is_subject = no has_capitulated = no NOT = { has_country_flag = HSM_VER_charter_suspended has_war_with = ERI has_war_with = EYR } }
			HSM_VER_apply_court = yes
			if = { limit = { has_country_flag = HSM_VER_staff_established NOT = { HSM_VER_SP_has_HSM_VER_joint_staff = yes } } HSM_VER_SP_add_HSM_VER_joint_staff = yes }
			if = { limit = { has_country_flag = HSM_VER_services_established NOT = { HSM_VER_SP_has_HSM_VER_shared_services = yes } } HSM_VER_SP_add_HSM_VER_shared_services = yes }
		}
		else = { HSM_VER_remove_charter_benefits = yes }
	}
	if = {
		limit = { tag = ERI }
		HSM_VER_reconcile_negotiation_timeouts = yes
		HSM_VER_migrate_negotiation_revision = yes
		# Renewed consent restores the completed institution, never the finale's rewards.
		if = {
			limit = { has_completed_focus = HSM_ERI_generation_without_feud HSM_VER_charter_valid = yes NOT = { has_country_flag = HSM_VER_union_established } }
			set_country_flag = HSM_VER_union_established
		}
		HSM_VER_queue_judgement = yes
	}
	if = { limit = { tag = AVI } HSM_VER_offer_avian_charter = yes HSM_VER_offer_avian_defence = yes }
}
