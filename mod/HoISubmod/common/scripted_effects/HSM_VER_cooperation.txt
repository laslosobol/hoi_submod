# The resolver writes only an uncommitted AI's existing route flag; human choices are never manufactured.
# Shared completion records are not an application for the next vacant presidency.
HSM_VER_COOP_release_shared_history = {
	ERI = {
		if = {
			limit = { exists = yes has_country_flag = HSM_VER_COOP_shared_history NOT = { has_country_flag = HSM_VER_UNI_leader has_country_flag = HSM_VER_UNI_founded HSM_VER_COOP_has_live_membership = yes } }
			uncomplete_national_focus = { focus = HSM_VER_UNI_convention }
			uncomplete_national_focus = { focus = HSM_VER_UNI_customs_union }
			uncomplete_national_focus = { focus = HSM_VER_UNI_common_works }
			uncomplete_national_focus = { focus = HSM_VER_UNI_common_administration }
			uncomplete_national_focus = { focus = HSM_VER_UNI_unified_command }
			clr_country_flag = HSM_VER_COOP_shared_history
		}
	}
	EYR = {
		if = {
			limit = { exists = yes has_country_flag = HSM_VER_COOP_shared_history NOT = { has_country_flag = HSM_VER_UNI_leader has_country_flag = HSM_VER_UNI_founded HSM_VER_COOP_has_live_membership = yes } }

HSM_VER_COOP_coordinate_routes = {
	ERI = {
		if = {
			limit = {
				exists = yes is_ai = yes is_subject = no
				HSM_VER_UNI_government = yes
				NOT = { HSM_VER_COOP_military_choice = yes HSM_VER_COOP_diplomatic_choice = yes }
			}
			if = {
				limit = { EYR = { exists = yes HSM_VER_COOP_military_choice = yes } }
				set_country_flag = HSM_VER_military_route
			}
			else_if = {
				limit = {
					HSM_VER_diplomatic_routes_open = yes
					EYR = { exists = yes is_subject = no HSM_VER_UNI_government = yes NOT = { HSM_VER_COOP_military_choice = yes } OR = { is_ai = yes HSM_VER_COOP_diplomatic_choice = yes } }
