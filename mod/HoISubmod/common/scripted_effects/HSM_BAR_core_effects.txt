# Maintenance is deliberately silent: callers expose rewards and unlocks, not the whole ledger.
HSM_BAR_core_refresh = {
 hidden_effect = {
  # Retired institutional meters must not survive as hidden progression gates.
  clear_variable = HSM_BAR_public_trust
  clear_variable = HSM_BAR_capacity
  # Remove obsolete secrecy data when continuing an existing campaign.
  clear_variable = HSM_BAR_cover_integrity
  clear_variable = HSM_BAR_service_coordination
  HSM_BAR_refresh_service_training = yes
  if = { limit = { has_country_flag = HSM_BAR_collegium } HSM_BAR_collegium_refresh = yes }
  HSM_BAR_refresh_case_outcomes = yes
 }
}
