HSM_BAR_cases_rock_complete = {
    if = {
        limit = { HSM_BAR_cases_rock_eligible = yes }
        set_country_flag = HSM_BAR_study_rock_done
        clr_country_flag = HSM_BAR_study_rock_retry
        country_event = { id = hsm_bar_cases.1 }
    }
    else_if = {
        limit = { tag = BAR NOT = { has_country_flag = HSM_BAR_study_rock_done } }
        set_country_flag = HSM_BAR_study_rock_retry
        HSM_BAR_case_failed = yes
    }
}
HSM_BAR_cases_rock_apply = {
    if = {
        limit = {
            tag = BAR has_country_flag = HSM_BAR_active has_country_flag = HSM_BAR_directorate has_country_flag = HSM_BAR_service_founded has_capitulated = no
            HSM_BAR_cases_actor = yes
            HSM_BAR_cases_rock_access = yes
            NOT = { has_political_power < 15 }
            has_country_flag = HSM_BAR_study_rock_done
            NOT = { has_country_flag = HSM_BAR_study_rock_applied has_country_flag = HSM_BAR_study_rock_reserved has_country_flag = HSM_BAR_case_rock_protect has_country_flag = HSM_BAR_case_rock_restrict }
        }
        set_country_flag = HSM_BAR_study_rock_applied
        set_country_flag = HSM_BAR_outcome_rock_recorded
        add_political_power = -15
        HSM_BAR_refresh_case_outcomes = yes
    }
}
