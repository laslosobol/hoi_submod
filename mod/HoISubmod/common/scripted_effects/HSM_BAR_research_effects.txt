# Public setup effect: country scope, called by the common institute focus.
# The facility check can be retried after liberation; bootstrap grants happen only once.
HSM_BAR_establish_institute = {
    if = {
        limit = {
            tag = BAR
            has_country_flag = HSM_BAR_active
        }
        set_country_flag = HSM_BAR_institute_open
        set_technology = { tech_magical_facilities = 1 popup = no }
        if = {
            limit = { has_dlc = "Gotterdammerung" }
            if = {
                limit = { NOT = { HSM_BAR_research_has_facility = yes } }
                if = {
                    limit = {
                        capital_scope = {
                            is_owned_by = PREV
                            is_fully_controlled_by = PREV
                            can_build_facility_here = yes
                            is_demilitarized_zone = no
                        }
                    }
                    capital_scope = {
                        construct_building_in_random_province = { magical_facility = 1 }
                    }
                }
                else = {
                    random_owned_controlled_state = {
                        limit = {
                            is_fully_controlled_by = PREV
                            can_build_facility_here = yes
                            is_demilitarized_zone = no
                        }
                        construct_building_in_random_province = { magical_facility = 1 }
                    }
                }
            }
            if = {
                limit = { NOT = { has_country_flag = HSM_BAR_research_bootstrap_granted } }
                set_country_flag = HSM_BAR_research_bootstrap_granted
                add_breakthrough_points = { specialization = specialization_magical value = 2 }
                generate_scientist_character = {
                    skills = { specialization_magical = 2 }
                }
                custom_effect_tooltip = HSM_BAR_research_scientist_tt
            }
        }
        custom_effect_tooltip = HSM_BAR_research_institute_tt
    }
}

        limit = {
            HSM_BAR_research_clinic_eligible = yes

            HSM_BAR_research_clinic_finish = yes
        }
        else_if = {
            limit = { NOT = { has_country_flag = HSM_BAR_research_clinic_choice_pending } }
            set_country_flag = HSM_BAR_research_clinic_choice_pending
            country_event = { id = HSM_BAR_research.1 }
        }
    }
}

    if = {
        limit = {
            HSM_BAR_research_clinic_eligible = yes
            has_country_flag = HSM_BAR_research_clinic_ready
            OR = { has_country_flag = HSM_BAR_research_clinic_consent has_country_flag = HSM_BAR_research_clinic_expedience }
        }
        set_country_flag = HSM_BAR_project_clinic_done
        clr_country_flag = HSM_BAR_research_clinic_ready
        clr_country_flag = HSM_BAR_research_clinic_choice_pending
        add_tech_bonus = {
            name = HSM_BAR_research_clinic_bonus
            bonus = 0.5
            uses = 1
            category = hospital_tech

        if = {
            limit = { has_country_flag = HSM_BAR_research_clinic_consent }
            add_political_power = -25
            add_stability = 0.03
            add_timed_idea = { idea = HSM_BAR_research_clinic_consent_idea days = 540 }

        else_if = {
            limit = { has_country_flag = HSM_BAR_research_clinic_expedience }
            add_political_power = -10
            add_stability = -0.04
            add_timed_idea = { idea = HSM_BAR_research_clinic_expedience_idea days = 365 }
