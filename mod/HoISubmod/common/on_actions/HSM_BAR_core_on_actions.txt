on_actions = {
 on_puppet = {
  effect = { if = { limit = { ROOT = { original_tag = BAR has_country_flag = HSM_BAR_active } } ROOT = { HSM_BAR_dissolve_institutions = yes } } }
 }
 on_release_as_free = {
  effect = { if = { limit = { ROOT = { original_tag = BAR has_country_flag = HSM_BAR_active } } ROOT = { HSM_BAR_dissolve_institutions = yes } } }
 }
 on_liberate = {
  effect = { if = { limit = { ROOT = { original_tag = BAR has_country_flag = HSM_BAR_active } } ROOT = { HSM_BAR_dissolve_institutions = yes } } }
 }
 on_annex = {
  effect = { if = { limit = { FROM = { original_tag = BAR has_country_flag = HSM_BAR_active } } FROM = { HSM_BAR_dissolve_institutions = yes } } }
 }
 on_daily_BAR = {
  effect = {
   if = { limit = { has_country_flag = HSM_BAR_service_gui_initialized } HSM_BAR_service_gui_pulse = yes }
   if = { limit = { has_country_flag = HSM_BAR_investigation_initialized } HSM_BAR_investigation_pulse = yes }
   if = { limit = { has_country_flag = HSM_BAR_fielddesk_initialized } HSM_BAR_fielddesk_pulse = yes }
   if = { limit = { has_country_flag = HSM_BAR_active } HSM_BAR_core_pulse = yes }
  }
 }
 on_daily_EQS = {
  effect = {
   if = {
    limit = {
     has_idea = HSM_BAR_anti_sombra
     NOT = { BAR = { exists = yes HSM_BAR_sombra_cooperation = yes } }
    }
    remove_ideas = HSM_BAR_anti_sombra
   }
  }
 }
}
