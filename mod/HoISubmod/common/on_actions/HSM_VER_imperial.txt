on_actions = {
	# ROOT annexes FROM. A released former host cannot reuse its old founding proof.
	on_annex = {
		effect = {
			FROM = {
				if = { limit = { has_country_flag = HSM_VER_UNI_founded } set_country_flag = HSM_VER_IMP_union_dispossessed }
			}
		}
	}
	# ROOT is declarer, FROM is the direct target. Joining someone else's war is insufficient.
	on_declare_war = {
		effect = {
			# A subsequent offensive charter war cannot reuse an earlier defensive defeat.
			if = {
				limit = { HSM_VER_IMP_actual_signatory = yes FROM = { HSM_VER_IMP_original_empire = yes } }
				ERI = { clr_country_flag = HSM_VER_IMP_defeat_recorded clr_country_flag = HSM_VER_IMP_peace_recorded }
			}
			if = {
				limit = {
					HSM_VER_IMP_original_empire = yes
					FROM = { HSM_VER_IMP_actual_signatory = yes }
					ERI = { HSM_VER_IMP_actual_signatory = yes }
				}
				set_country_flag = HSM_VER_IMP_recorded_aggressor
				FROM = { set_country_flag = HSM_VER_IMP_attacked_signatory }
				ERI = {
					set_country_flag = HSM_VER_IMP_defensive_war_recorded
					clr_country_flag = HSM_VER_IMP_defeat_recorded
					clr_country_flag = HSM_VER_IMP_peace_recorded
				}
			}
		}
	}

	# ROOT is loser, FROM is victor. Both ERI and the recorded victim must still defend.
	on_capitulation_immediate = {
		effect = {
			if = {
				limit = {
					HSM_VER_IMP_original_empire = yes
					has_country_flag = HSM_VER_IMP_recorded_aggressor
					ERI = {
						HSM_VER_IMP_actual_signatory = yes
						has_country_flag = HSM_VER_IMP_defensive_war_recorded
						has_defensive_war_with = ROOT
						NOT = { has_offensive_war_with = ROOT }
					}
					FROM = {
						has_defensive_war_with = ROOT
						NOT = { has_offensive_war_with = ROOT }
						OR = { tag = ERI has_war_together_with = ERI }
					}
					any_country = {
						HSM_VER_IMP_actual_signatory = yes
						has_country_flag = HSM_VER_IMP_attacked_signatory
						has_defensive_war_with = ROOT
						NOT = { has_offensive_war_with = ROOT }
					}
				}
				ERI = { set_country_flag = HSM_VER_IMP_defeat_recorded }
			}
		}
	}

	on_uncapitulation = {
		effect = {
			if = {
				limit = { HSM_VER_IMP_original_empire = yes ERI = { has_war_with = ROOT } }
				ERI = { clr_country_flag = HSM_VER_IMP_defeat_recorded clr_country_flag = HSM_VER_IMP_peace_recorded }
			}
		}
	}

	# ROOT is winner, FROM is loser; a third-party peace conference cannot certify ERI.
	on_peaceconference_ended = {
		effect = {
			if = {
				limit = {
					tag = ERI
					original_tag = ERI
					has_country_flag = HSM_VER_IMP_defensive_war_recorded
					has_country_flag = HSM_VER_IMP_defeat_recorded
					NOT = { has_country_flag = bypass_on_peaceconference_ended }
					FROM = {
						HSM_VER_IMP_original_empire = yes
						has_country_flag = HSM_VER_IMP_recorded_aggressor
						NOT = { has_country_flag = bypass_on_peaceconference_ended }
					}
				}
				set_country_flag = HSM_VER_IMP_peace_recorded
			}
		}
	}

	on_daily_ERI = {
		effect = {
			ERI = {
				# Forget a victim after its defensive war ends. Flags are never backfilled from a save.
				if = { limit = { NOT = { has_defensive_war_with = GRI } } clr_country_flag = HSM_VER_IMP_attacked_signatory }
				EYR = { if = { limit = { NOT = { has_defensive_war_with = GRI } } clr_country_flag = HSM_VER_IMP_attacked_signatory } }
				AVI = { if = { limit = { NOT = { has_defensive_war_with = GRI } } clr_country_flag = HSM_VER_IMP_attacked_signatory } }
				if = {
					limit = { NOT = { has_war_with = GRI } NOT = { has_country_flag = HSM_VER_IMP_peace_recorded } }
					clr_country_flag = HSM_VER_IMP_defeat_recorded
				}
				if = {
					limit = {
						NOT = { has_war_with = GRI }
						NOT = { any_country = { HSM_VER_IMP_actual_signatory = yes has_country_flag = HSM_VER_IMP_attacked_signatory } }
					}
					GRI = { clr_country_flag = HSM_VER_IMP_recorded_aggressor }
					if = {
						limit = { NOT = { has_country_flag = HSM_VER_IMP_peace_recorded } }
						clr_country_flag = HSM_VER_IMP_defensive_war_recorded
						clr_country_flag = HSM_VER_IMP_defeat_recorded
					}
				}
				HSM_VER_IMP_check_mandate = yes
				HSM_VER_IMP_queue_internal_protocol = yes
			}
		}
	}
}
