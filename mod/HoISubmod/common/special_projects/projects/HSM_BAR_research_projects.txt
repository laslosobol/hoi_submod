    specialization = specialization_magical
    icon = GFX_sp_magical_thaumaturgical_diagnoses
    project_tags = { sp_tag_magical }
    allowed = { original_tag = BAR has_dlc = "Gotterdammerung" }
    visible = {
        FROM = { HSM_BAR_service_legacy_visible = yes
            tag = BAR
            has_country_flag = HSM_BAR_active
            has_country_flag = HSM_BAR_institute_open
        }
    }
    available = { FROM = { HSM_BAR_research_clinic_can_start = yes } }
    resource_cost = { resources = { crystals = 2 steel = 2 } }
    complexity = sp_complexity.small
    prototype_time = 30

        base = 1
        modifier = {
            factor = 0
            FROM = { NOT = { OR = { has_country_flag = HSM_BAR_research_clinic_consent has_country_flag = HSM_BAR_research_clinic_expedience } } }
        }
    }
    project_output = {
        country_effects = { HSM_BAR_research_clinic_complete = yes }
    }
    unique_prototype_rewards = {

                    country_effects = { HSM_BAR_research_clinic_choose_consent = yes }

            option = {
