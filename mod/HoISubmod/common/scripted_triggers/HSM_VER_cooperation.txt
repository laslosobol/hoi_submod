# Joint membership is explicit consent to the current presidency, never merely a tag.

HSM_VER_COOP_has_live_membership = {
	OR = {
		AND = { has_country_flag = HSM_VER_UNI_member_ERI ERI = { exists = yes has_country_flag = HSM_VER_UNI_leader NOT = { has_country_flag = HSM_VER_UNI_closed has_country_flag = HSM_VER_UNI_founded } } }
		AND = { has_country_flag = HSM_VER_UNI_member_EYR EYR = { exists = yes has_country_flag = HSM_VER_UNI_leader NOT = { has_country_flag = HSM_VER_UNI_closed has_country_flag = HSM_VER_UNI_founded } } }
		AND = { has_country_flag = HSM_VER_UNI_member_AVI AVI = { exists = yes has_country_flag = HSM_VER_UNI_leader NOT = { has_country_flag = HSM_VER_UNI_closed has_country_flag = HSM_VER_UNI_founded } } }
	}
}

# Started military focuses protect a player's choice before the completion flag exists.
HSM_VER_COOP_no_military_choice = {
	ERI = { NOT = { HSM_VER_COOP_military_choice = yes } }
	EYR = { NOT = { HSM_VER_COOP_military_choice = yes } }
}
HSM_VER_COOP_military_choice = {
	OR = {
		has_country_flag = HSM_VER_military_route
		AND = { tag = ERI OR = { has_completed_focus = HSM_ERI_FORCE_the_price_of_victory focus_progress = { focus = HSM_ERI_FORCE_the_price_of_victory progress > 0 } } }
		AND = { tag = EYR OR = { has_completed_focus = HSM_EYR_FORCE_the_last_campaign focus_progress = { focus = HSM_EYR_FORCE_the_last_campaign progress > 0 } } }
	}
}
