# Country scope throughout; project availability must enter FROM first.
HSM_BAR_research_actor = {
    tag = BAR
    has_country_flag = HSM_BAR_active
    has_country_flag = HSM_BAR_institute_open
    has_capitulated = no
    OR = {
        has_country_flag = HSM_BAR_collegium
        AND = { has_country_flag = HSM_BAR_directorate has_country_flag = HSM_BAR_service_founded }
    }
}

HSM_BAR_research_has_facility = {
    any_owned_state = {
        is_fully_controlled_by = PREV
        magical_facility > 0
    }
}

HSM_BAR_research_clinic_eligible = {
    custom_trigger_tooltip = {
        tooltip = HSM_BAR_research_clinic_eligible_tt
        HSM_BAR_research_actor = yes
        has_country_flag = HSM_BAR_collegium
        NOT = { has_country_flag = HSM_BAR_project_clinic_done }
    }
}

# Suspension blocks new trials, not ratification of earned clinical results.
HSM_BAR_research_clinic_can_start = {
    HSM_BAR_research_clinic_eligible = yes
    custom_trigger_tooltip = {
        tooltip = HSM_BAR_research_clinic_trials_open_tt
        NOT = { has_country_flag = HSM_BAR_collegium_trial_pause }
    }
}
