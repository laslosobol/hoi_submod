# All five original ducal provinces must have a settled owner, not just an occupier.
# Explicit tags keep this valid when tested in FROM during Avian's response.
HSM_VER_FORCE_settlement_held = {
	custom_trigger_tooltip = {
		tooltip = HSM_VER_FORCE_settlement_tt
		HSM_VER_FORCE_eligible_house = yes
		HSM_VER_FORCE_at_war = no
		has_country_flag = HSM_VER_FORCE_war_recorded
		has_country_flag = HSM_VER_FORCE_rival_defeated
		has_country_flag = HSM_VER_FORCE_peace_recorded
		NOT = { has_country_flag = HSM_VER_FORCE_settling }
		OR = {
			AND = {
				tag = ERI owns_state = 401 owns_state = 500 owns_state = 673
				401 = { is_fully_controlled_by = ERI } 500 = { is_fully_controlled_by = ERI } 673 = { is_fully_controlled_by = ERI }
				OR = { EYR = { exists = no } EYR = { is_subject_of = ERI has_capitulated = no } }
				391 = { OR = { AND = { is_owned_by = ERI is_fully_controlled_by = ERI } AND = { is_owned_by = EYR is_fully_controlled_by = EYR EYR = { is_subject_of = ERI } } } }
				400 = { OR = { AND = { is_owned_by = ERI is_fully_controlled_by = ERI } AND = { is_owned_by = EYR is_fully_controlled_by = EYR EYR = { is_subject_of = ERI } } } }
			}
			AND = {
				tag = EYR owns_state = 391 owns_state = 400
				391 = { is_fully_controlled_by = EYR } 400 = { is_fully_controlled_by = EYR }
				OR = { ERI = { exists = no } ERI = { is_subject_of = EYR has_capitulated = no } }
				401 = { OR = { AND = { is_owned_by = EYR is_fully_controlled_by = EYR } AND = { is_owned_by = ERI is_fully_controlled_by = ERI ERI = { is_subject_of = EYR } } } }
				500 = { OR = { AND = { is_owned_by = EYR is_fully_controlled_by = EYR } AND = { is_owned_by = ERI is_fully_controlled_by = ERI ERI = { is_subject_of = EYR } } } }
				673 = { OR = { AND = { is_owned_by = EYR is_fully_controlled_by = EYR } AND = { is_owned_by = ERI is_fully_controlled_by = ERI ERI = { is_subject_of = EYR } } } }
			}
		}
	}
}
