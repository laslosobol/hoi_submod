# A museum/seal stays at its recorded physical site even if its original country
# disappears. A worn crown is never an extraction target. No leader kidnapping.
HSM_BAR_case_crown_at_target = {
    has_country_flag = HSM_BAR_case_crown_exists
    NOT = { has_country_flag = HSM_BAR_case_crown_missing }
    has_variable = HSM_BAR_case_crown_site
    var:HSM_BAR_case_crown_site = { is_controlled_by = FROM }
}
HSM_BAR_case_crown_extractable = {
    HSM_BAR_case_crown_at_target = yes
    has_country_flag = HSM_BAR_case_crown_known
    has_country_flag = HSM_BAR_case_crown_investigated
    OR = {
        has_country_flag = HSM_BAR_case_crown_museum
        has_country_flag = HSM_BAR_case_crown_sealed
    }
    NOT = { has_country_flag = HSM_BAR_case_crown_leader }
}
