HSM_CYA_displaced_court_available = {
	tag = CYA
	has_country_flag = HSM_CYA_path_imperial_administration
	NOT = { has_country_flag = HSM_CYA_exile_court_resolved }
	is_subject = no
	has_capitulated = no
	date < 1021.5.21
	NOT = { has_global_flag = GRI_grover_vi_dead }
	NOT = { HSM_CYA_grover_reigns_anywhere = yes }
	NOT = { has_country_flag = HSM_CYA_grover_reigned_elsewhere }
	NOT = { has_character = HSM_CYA_ferdinand_dawnclaw }
	has_country_leader = { character = CYA_countess_taillow_sumpfkiel ruling_only = yes }
	NOT = { has_war_with = GRI }
	GRI = {
		has_character = GRI_emperor_grover_vi
		OR = { exists = no has_capitulated = yes }
		OR = {
			AND = {
				has_character = GRI_archon_eros_vii
				has_country_leader = { character = GRI_archon_eros_vii ruling_only = yes }
				NOT = { has_global_flag = GRI_eros_dead }
			}
			AND = {
				has_character = STW_gabriela_eagleclaw
				has_country_leader = { character = STW_gabriela_eagleclaw ruling_only = yes }
			}
		}
	}
	382 = {
		owner = {
			has_country_flag = holds_griffon_capital
			controls_state = 382
			NOT = { tag = GRI }
			NOT = { tag = ROOT }
			NOT = { tag = BRZ }
			NOT = { has_idea = GRI_grover_vi }
			NOT = { has_character = GRI_emperor_grover_vi }
			NOT = { has_focus_tree = angriver_focus_blackclaw_imperial }
			NOT = { has_country_leader = { character = ANG_baron_leer_the_vicious ruling_only = yes } }
			NOT = { AND = { tag = PYT OR = { has_government = fascism has_government = communism } } }
			NOT = {
				AND = {
					tag = YAL
					OR = {
						has_government = fascism
						has_government = communism
						has_country_flag = YAL_right_coalition_chosen
						has_country_leader = { character = YAL_emperor_grover_ii ruling_only = yes }
					}
				}
			}
		}
	}
}

# The base mod records original_cores for each tag, including an annexed country.
HSM_CYA_original_griffonian_cores_secured = {
	all_of_scopes = {
		array = original_cores
		OR = {
			NOT = { is_on_continent = asia }
			HSM_CYA_state_in_imperial_sphere = yes
		}
	}
}

# State scope: annexation by an unrelated third party is not imperial unification.
HSM_CYA_state_in_imperial_sphere = {
	owner = { OR = { tag = ROOT is_subject_of = ROOT } }
	controller = { OR = { tag = ROOT is_subject_of = ROOT } }
}
